5   package com.qulice.errorprone;
6   
7   import com.qulice.spi.Environment;
8   import com.qulice.spi.Violation;
9   import java.io.File;
10  import java.io.IOException;
11  import java.util.ArrayList;
12  import java.util.Collection;
13  import java.util.Collections;
14  import org.hamcrest.MatcherAssert;
15  import org.hamcrest.Matchers;
16  import org.junit.jupiter.api.Test;
17  
18  /**
19   * Test case for {@link ErrorProneValidator}'s source-level awareness of the
20   * stock {@code PatternMatchingInstanceof} check. The check suggests
21   * pattern-matching {@code instanceof}, a Java 16+ feature, so it must not fire
22   * when the project compiles at an older source level (issue #1716).
23   * @since 1.0
24   */
25  final class ErrorPronePatternMatchingInstanceofTest {
26  
27      /**
28       * Name of the ErrorProne check under test.
29       */
30      private static final String CHECK = "PatternMatchingInstanceof";
31  
32      /**
33       * Path of the sample file, relative to the base directory.
34       */
35      private static final String FILE = "src/main/java/foo/Sample.java";
36  
37      /**
38       * A class whose {@code instanceof} followed by a cast to the same type is
39       * exactly what {@code PatternMatchingInstanceof} suggests rewriting.
40       */
41      private static final String SOURCE = String.join(
42          System.lineSeparator(),
43          "package foo;",
44          "/**",
45          " * Sample.",
46          " * @since 1.0",
47          " */",
48          "public final class Sample {",
49          "    /**",
50          "     * Compare with another object.",
51          "     * @param other The other object",
52          "     * @return TRUE if the same",
53          "     */",
54          "    public boolean same(final Object other) {",
55          "        return other instanceof Sample",
56          "            && ((Sample) other).hashCode() == this.hashCode();",
57          "    }",
58          "}"
59      );
60  
61      @Test
62      void skipsPatternMatchingInstanceofOnSourceEight() throws Exception {
63          MatcherAssert.assertThat(
64              "PatternMatchingInstanceof must not fire when source is 8",
65              this.checks("8"),
66              Matchers.not(Matchers.hasItem(ErrorPronePatternMatchingInstanceofTest.CHECK))
67          );
68      }
69  
70      @Test
71      void skipsPatternMatchingInstanceofOnLegacySourceEight() throws Exception {
72          MatcherAssert.assertThat(
73              "PatternMatchingInstanceof must not fire when source is 1.8",
74              this.checks("1.8"),
75              Matchers.not(Matchers.hasItem(ErrorPronePatternMatchingInstanceofTest.CHECK))
76          );
77      }
78  
79      @Test
80      void reportsPatternMatchingInstanceofOnModernSource() throws Exception {
81          MatcherAssert.assertThat(
82              "PatternMatchingInstanceof must fire when source is 17",
83              this.checks("17"),
84              Matchers.hasItem(ErrorPronePatternMatchingInstanceofTest.CHECK)
85          );
86      }
87  
88      private Collection<String> checks(final String source) throws IOException {
89          final Environment.Mock mock = new Environment.Mock();
90          if (source != null) {
91              mock.withParam("maven.compiler.source", source);
92          }
93          final Environment env = mock.withFile(
94              ErrorPronePatternMatchingInstanceofTest.FILE,
95              ErrorPronePatternMatchingInstanceofTest.SOURCE
96          );
97          final Collection<Violation> violations = new ErrorProneValidator(env).validate(
98              Collections.singletonList(
99                  new File(env.basedir(), ErrorPronePatternMatchingInstanceofTest.FILE)
100             )
101         );
102         final Collection<String> names = new ArrayList<>(violations.size());
103         for (final Violation violation : violations) {
104             names.add(violation.name());
105         }
106         return names;
107     }
108 }
