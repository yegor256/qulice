5   package com.qulice.errorprone;
6   
7   import org.hamcrest.MatcherAssert;
8   import org.hamcrest.Matchers;
9   import org.junit.jupiter.api.Assertions;
10  import org.junit.jupiter.api.Test;
11  
12  /**
13   * Test case for {@link Xplugin}.
14   * @since 1.0
15   */
16  final class XpluginTest {
17  
18      @Test
19      void switchesQuliceOwnPatternsOff() {
20          MatcherAssert.assertThat(
21              "the patterns Qulice disables must always be disabled",
22              new Xplugin("").argument(),
23              Matchers.equalTo(
24                  String.join(
25                      " ",
26                      "-Xplugin:ErrorProne",
27                      "-Xep:InvalidBlockTag:OFF",
28                      "-Xep:OperatorPrecedence:OFF",
29                      "-Xep:UnicodeInCode:OFF"
30                  )
31              )
32          );
33      }
34  
35      @Test
36      void appendsFlagsOfProjectLast() {
37          MatcherAssert.assertThat(
38              "the project must have the last word on every pattern",
39              new Xplugin("-Xep:OperatorPrecedence:ERROR").argument(),
40              Matchers.endsWith(
41                  "-Xep:UnicodeInCode:OFF -Xep:OperatorPrecedence:ERROR"
42              )
43          );
44      }
45  
46      @Test
47      void takesFlagsSeparatedByCommas() {
48          MatcherAssert.assertThat(
49              "commas must separate flags just like spaces do",
50              new Xplugin(" -Xep:UnusedVariable:OFF, -Xep:SelfAssignment:OFF ")
51                  .argument(),
52              Matchers.endsWith(
53                  "-Xep:UnusedVariable:OFF -Xep:SelfAssignment:OFF"
54              )
55          );
56      }
57  
58      @Test
59      void countsPatternsThatFire() {
60          MatcherAssert.assertThat(
61              "ErrorProne cannot stay silent about how many patterns it applies",
62              new Xplugin("").patterns(),
63              Matchers.greaterThan(100)
64          );
65      }
66  
67      @Test
68      void countsOnePatternFewerWhenProjectSwitchesOneOff() {
69          MatcherAssert.assertThat(
70              "a pattern the project switches off must leave the count",
71              new Xplugin("-Xep:UnusedVariable:OFF").patterns(),
72              Matchers.equalTo(new Xplugin("").patterns() - 1)
73          );
74      }
75  
76      @Test
77      void countsOnePatternMoreWhenProjectSwitchesOneBackOn() {
78          MatcherAssert.assertThat(
79              "a pattern the project switches back on must join the count",
80              new Xplugin("-Xep:OperatorPrecedence:ERROR").patterns(),
81              Matchers.equalTo(new Xplugin("").patterns() + 1)
82          );
83      }
84  
85      @Test
86      void refusesFlagErrorProneWouldNotUnderstand() {
87          MatcherAssert.assertThat(
88              "the flag that cannot work must be named in the failure",
89              Assertions.assertThrows(
90                  IllegalArgumentException.class,
91                  () -> new Xplugin("-Xlint:all").argument(),
92                  "a flag that is not an ErrorProne one must be refused"
93              ).getMessage(),
94              Matchers.containsString("-Xlint:all")
95          );
96      }
97  }
