5   package com.qulice.maven;
6   
7   import com.qulice.spi.Environment;
8   import com.qulice.spi.Validator;
9   import java.util.concurrent.atomic.AtomicReference;
10  
11  /**
12   * A test fake {@link Validator} that remembers the value one
13   * {@link Environment} parameter had when the validator was called.
14   * @since 1.0
15   */
16  final class RecordingValidator implements Validator {
17  
18      /**
19       * Name of the parameter to remember.
20       */
21      private final String param;
22  
23      /**
24       * What the parameter was, or empty if the validator was never called.
25       */
26      private final AtomicReference<String> value;
27  
28      RecordingValidator(final String name) {
29          this.param = name;
30          this.value = new AtomicReference<>("");
31      }
32  
33      @Override
34      public void validate(final Environment env) {
35          this.value.set(env.param(this.param, ""));
36      }
37  
38      @Override
39      public String name() {
40          return "Recording";
41      }
42  
43      String seen() {
44          return this.value.get();
45      }
46  }
