215     /**
216      * DefaultMavenEnvironment.files() should silently drop binary files so
217      * that validators never try to read them as text
218      * (see <a href="https://github.com/yegor256/qulice/issues/1264">
219      * issue #1264</a>).
220      *
221      * @param basedir Temporary base directory
222      * @throws Exception If something wrong happens inside
223      */
224     @Test
225     void skipsBinaryFilesWhenListing(@TempDir final Path basedir)
226         throws Exception {
227         final Path src = basedir.resolve("src/main/java");
228         Files.createDirectories(src);
229         final Path source = src.resolve("Foo.java");
230         Files.writeString(
231             source,
232             "class Foo {}".concat(String.valueOf('\n')),
233             StandardCharsets.UTF_8
234         );
235         final Path image = basedir.resolve("src/main/resources/pixel.png");
236         Files.createDirectories(image.getParent());
237         Files.write(
238             image,
239             new byte[] {
240                 (byte) 0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A,
241                 0x00, 0x00, 0x00, 0x0D, 0x49, 0x48, 0x44, 0x52,
242             }
243         );
244         final DefaultMavenEnvironment env = new DefaultMavenEnvironment();
245         env.setProject(
246             new MavenProjectStub() {
247                 @Override
248                 public File getBasedir() {
249                     return basedir.toFile();
250                 }
251             }
252         );
253         MatcherAssert.assertThat(
254             "Binary files cannot leak into the list of files to validate",
255             env.files("*.*"),
256             Matchers.allOf(
257                 Matchers.hasItem(source.toFile()),
258                 Matchers.not(Matchers.hasItem(image.toFile()))
259             )
260         );
261     }

584     /**
585      * DefaultMavenEnvironment.tempdir() must not point at the directory
586      * with compiled classes, otherwise validators drop their scratch files
587      * into {@code target/classes} and they end up in the JAR and confuse
588      * {@code jacoco:report}
589      * (see <a href="https://github.com/yegor256/qulice/issues/1741">
590      * issue #1741</a>).
591      *
592      * @param basedir Temporary base directory
593      */
594     @Test
595     void keepsTempdirOutsideOutdir(@TempDir final Path basedir) {
596         final Path target = basedir.resolve("target");
597         final Build build = new Build();
598         build.setDirectory(target.toAbsolutePath().toString());
599         build.setOutputDirectory(
600             target.resolve("classes").toAbsolutePath().toString()
601         );
602         final DefaultMavenEnvironment env = new DefaultMavenEnvironment();
603         env.setProject(
604             new MavenProjectStub() {
605                 @Override
606                 public File getBasedir() {
607                     return basedir.toFile();
608                 }
609 
610                 @Override
611                 public Build getBuild() {
612                     return build;
613                 }
614             }
615         );
616         final Path temp = env.tempdir().toPath().toAbsolutePath();
617         MatcherAssert.assertThat(
618             "Scratch dir must be usable, under target/ and away from classes",
619             env.tempdir().isDirectory()
620                 && temp.startsWith(target.toAbsolutePath())
621                 && !temp.startsWith(env.outdir().toPath().toAbsolutePath()),
622             Matchers.is(true)
623         );
624     }
