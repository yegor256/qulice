5   package com.qulice.maven;
6   
7   import com.qulice.spi.ResourceValidator;
8   import com.qulice.spi.Validator;
9   import java.util.HashSet;
10  import java.util.Set;
11  
12  /**
13   * Mocker of ValidatorsProvider.
14   * @since 0.4
15   */
16  final class ValidatorsProviderMocker {
17  
18      /**
19       * List of external validators.
20       */
21      private final Set<Validator> external = new HashSet<>();
22  
23      /**
24       * List of external resource validators.
25       */
26      private final transient Set<ResourceValidator> rexternal = new HashSet<>();
27  
28      /**
29       * List of internal validators.
30       */
31      private final Set<MavenValidator> internal = new HashSet<>();
32  
33      /**
34       * With this external validator.
35       * @param validator The validator
36       * @return This object
37       */
38      ValidatorsProviderMocker withExternal(final Validator validator) {
39          this.external.add(validator);
40          return this;
41      }
42  
43      /**
44       * With this external resource validator.
45       * @param validator The validator
46       * @return This object
47       */
48      ValidatorsProviderMocker withExternalResource(final ResourceValidator validator) {
49          this.rexternal.add(validator);
50          return this;
51      }
52  
53      /**
54       * With this external validator.
55       * @param validator The validator
56       * @return This object
57       */
58      ValidatorsProviderMocker withInternal(final MavenValidator validator) {
59          this.internal.add(validator);
60          return this;
61      }
62  
63      /**
64       * Mock it.
65       * @return The provider
66       */
67      ValidatorsProvider mock() {
68          return new FakeValidatorsProvider(
69              this.internal,
70              this.external,
71              this.rexternal
72          );
73      }
74  }
