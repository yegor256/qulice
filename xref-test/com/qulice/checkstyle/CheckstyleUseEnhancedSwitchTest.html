20  /**
21   * Test case for {@link CheckstyleValidator}'s source-level awareness of the
22   * stock {@code UseEnhancedSwitch} check. The check suggests arrow-switch
23   * syntax, a Java 14+ feature, so it must be disabled when the project targets
24   * an older Java (issue #1694).
25   * @since 1.0
26   */
27  final class CheckstyleUseEnhancedSwitchTest {
28  
29      /**
30       * Name of the resource holding a convertible classic switch.
31       */
32      private static final String FILE = "EnhancedSwitch.java";
33  
34      /**
35       * Simple name of the stock check under test.
36       */
37      private static final String CHECK = "UseEnhancedSwitchCheck";
38  
39      @Test
40      void skipsEnhancedSwitchOnSourceEight() throws Exception {
41          MatcherAssert.assertThat(
42              "UseEnhancedSwitch must not fire when source is 8",
43              this.runValidation("8"),
44              Matchers.not(
45                  Matchers.hasItem(
46                      new ViolationMatcher(
47                          "", CheckstyleUseEnhancedSwitchTest.FILE, "",
48                          CheckstyleUseEnhancedSwitchTest.CHECK
49                      )
50                  )
51              )
52          );
53      }
54  
55      @Test
56      void skipsEnhancedSwitchOnLegacySourceEight() throws Exception {
57          MatcherAssert.assertThat(
58              "UseEnhancedSwitch must not fire when source is 1.8",
59              this.runValidation("1.8"),
60              Matchers.not(
61                  Matchers.hasItem(
62                      new ViolationMatcher(
63                          "", CheckstyleUseEnhancedSwitchTest.FILE, "",
64                          CheckstyleUseEnhancedSwitchTest.CHECK
65                      )
66                  )
67              )
68          );
69      }
70  
71      @Test
72      void skipsEnhancedSwitchWhenSourceUnknown() throws Exception {
73          MatcherAssert.assertThat(
74              "UseEnhancedSwitch must not fire when source is unknown",
75              this.runValidation(null),
76              Matchers.not(
77                  Matchers.hasItem(
78                      new ViolationMatcher(
79                          "", CheckstyleUseEnhancedSwitchTest.FILE, "",
80                          CheckstyleUseEnhancedSwitchTest.CHECK
81                      )
82                  )
83              )
84          );
85      }
86  
87      @Test
88      void reportsEnhancedSwitchOnModernSource() throws Exception {
89          MatcherAssert.assertThat(
90              "UseEnhancedSwitch must fire when source is 17",
91              this.runValidation("17"),
92              Matchers.hasItem(
93                  new ViolationMatcher(
94                      "", CheckstyleUseEnhancedSwitchTest.FILE, "",
95                      CheckstyleUseEnhancedSwitchTest.CHECK
96                  )
97              )
98          );
99      }
100 
101     /**
102      * Run the validator over the sample file with the given source level.
103      * @param source Value for {@code maven.compiler.source}, or {@code null}
104      * @return Collection of violations
105      * @throws IOException If some IO problem
106      */
107     private Collection<Violation> runValidation(final String source)
108         throws IOException {
109         final Environment.Mock mock = new Environment.Mock();
110         mock.withParam(
111             "license",
112             String.format(
113                 "file:%s",
114                 new License().savePackageInfo(
115                     new File(mock.basedir(), "src/main/java/foo")
116                 ).withLines("Hello.")
117                     .withEol(String.valueOf('\n')).file()
118             )
119         );
120         if (source != null) {
121             mock.withParam("maven.compiler.source", source);
122         }
123         final Environment env = mock.withFile(
124             String.format(
125                 "src/main/java/foo/%s",
126                 CheckstyleUseEnhancedSwitchTest.FILE
127             ),
128             new IoCheckedText(
129                 new TextOf(
130                     new ResourceOf(
131                         new FormattedText(
132                             "com/qulice/checkstyle/%s",
133                             CheckstyleUseEnhancedSwitchTest.FILE
134                         )
135                     )
136                 )
137             ).asString()
138         );
139         return new CheckstyleValidator(env).validate(
140             env.files(CheckstyleUseEnhancedSwitchTest.FILE)
141         );
142     }
