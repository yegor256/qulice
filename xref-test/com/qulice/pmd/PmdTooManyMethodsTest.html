5   package com.qulice.pmd;
6   
7   import org.hamcrest.Matchers;
8   import org.junit.jupiter.api.Test;
9   import org.junit.jupiter.params.ParameterizedTest;
10  import org.junit.jupiter.params.provider.ValueSource;
11  
12  /**
13   * Test case for {@link PmdValidator}'s handling of the
14   * {@code TooManyMethods} rule, which is disabled for test
15   * classes so that splitting asserts into separate {@code @Test}
16   * methods (required by {@code UnitTestContainsTooManyAsserts})
17   * does not push the class past the threshold, and which counts
18   * only public and protected methods.
19   * Regression test for https://github.com/yegor256/qulice/issues/1605,
20   * https://github.com/yegor256/qulice/issues/1647
21   * and https://github.com/yegor256/qulice/issues/1656
22   * @since 1.0
23   */
24  final class PmdTooManyMethodsTest {
25  
26      @ParameterizedTest
27      @ValueSource(
28          strings = {
29              "ManyMethodsTest.java",
30              "ManyMethodsIT.java",
31              "ManyMethodsNestedTest.java",
32              "ManyMethodsChecks.java",
33              "ManyPublicMethodsTest.java"
34          }
35      )
36      void allowsManyMethodsInTestClass(final String file) throws Exception {
37          new PmdAssert(
38              file,
39              Matchers.any(Boolean.class),
40              Matchers.not(
41                  Matchers.containsString("TooManyMethods")
42              )
43          ).assertOk();
44      }
45  
46      @Test
47      void reportsTooManyMethodsInNonTestClass() throws Exception {
48          new PmdAssert(
49              "ManyMethods.java",
50              Matchers.is(false),
51              Matchers.containsString("TooManyMethods")
52          ).assertOk();
53      }
54  
55      @Test
56      void reportsTooManyProtectedMethods() throws Exception {
57          new PmdAssert(
58              "ManyProtectedMethods.java",
59              Matchers.is(false),
60              Matchers.containsString("TooManyMethods")
61          ).assertOk();
62      }
63  
64      @Test
65      void ignoresPrivateMethods() throws Exception {
66          new PmdAssert(
67              "ManyPrivateMethods.java",
68              Matchers.any(Boolean.class),
69              Matchers.not(
70                  Matchers.containsString("TooManyMethods")
71              )
72          ).assertOk();
73      }
74  
75      @Test
76      void reportsRedundantSuppressionInTestClass() throws Exception {
77          new PmdAssert(
78              "ManyMethodsSuppressedTest.java",
79              Matchers.is(false),
80              Matchers.containsString("UnnecessaryWarningSuppression")
81          ).assertOk();
82      }
83  }
