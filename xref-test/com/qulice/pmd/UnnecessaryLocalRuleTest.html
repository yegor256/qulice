5   package com.qulice.pmd;
6   
7   import com.qulice.spi.Environment;
8   import com.qulice.spi.Violation;
9   import java.io.File;
10  import java.util.Collections;
11  import java.util.List;
12  import java.util.stream.Collectors;
13  import org.cactoos.text.TextOf;
14  import org.hamcrest.MatcherAssert;
15  import org.hamcrest.Matchers;
16  import org.junit.jupiter.api.Test;
17  
18  /**
19   * Test case for UnnecessaryLocalRule.
20   * @since 0.24
21   */
22  final class UnnecessaryLocalRuleTest {
23  
24      /**
25       * The name of the rule under test.
26       */
27      private static final String RULE = "UnnecessaryLocalRule";
28  
29      /**
30       * Path template for the fixture file in the mock environment.
31       */
32      private static final String PATH = "src/main/java/foo/%s";
33  
34      @Test
35      void detectsUnnecessaryLocalVariableOnReturn() throws Exception {
36          MatcherAssert.assertThat(
37              "UnnecessaryLocalRule should fire on a local variable used only in return",
38              this.violations("UnnecessaryLocal.java"),
39              Matchers.hasItem(UnnecessaryLocalRuleTest.RULE)
40          );
41      }
42  
43      @Test
44      void doesNotFireWhenVariableIsUsedInLoop() throws Exception {
45          MatcherAssert.assertThat(
46              "UnnecessaryLocalRule should not fire when variable is used in a loop",
47              this.violations("UnnecessaryLocalInLoop.java"),
48              Matchers.not(Matchers.hasItem(UnnecessaryLocalRuleTest.RULE))
49          );
50      }
51  
52      @Test
53      void doesNotFireWhenVariableIsUsedMoreThanOnce() throws Exception {
54          MatcherAssert.assertThat(
55              "UnnecessaryLocalRule should not fire when variable is used more than once",
56              this.violations("UnnecessaryLocalUsedTwice.java"),
57              Matchers.not(Matchers.hasItem(UnnecessaryLocalRuleTest.RULE))
58          );
59      }
60  
61      @Test
62      void doesNotFireWhenUseIsAcrossAnonymousClassOrLambda() throws Exception {
63          MatcherAssert.assertThat(
64              "UnnecessaryLocalRule should not fire when the only use is inside an anonymous class or lambda body",
65              this.violations("UnnecessaryLocalAcrossAnonymousClass.java"),
66              Matchers.not(Matchers.hasItem(UnnecessaryLocalRuleTest.RULE))
67          );
68      }
69  
70      @Test
71      void doesNotFireWhenInitialiserCapturesFreshState() throws Exception {
72          MatcherAssert.assertThat(
73              "UnnecessaryLocalRule should not fire when the initialiser captures a clock or randomness source",
74              this.violations("UnnecessaryLocalCapturedTimestamp.java"),
75              Matchers.not(Matchers.hasItem(UnnecessaryLocalRuleTest.RULE))
76          );
77      }
78  
79      @Test
80      void doesNotFireWhenInterveningCallMutatesSameTarget() throws Exception {
81          MatcherAssert.assertThat(
82              "UnnecessaryLocalRule should not fire when a later statement calls the same target",
83              this.violations("UnnecessaryLocalDestructiveCleanup.java"),
84              Matchers.not(Matchers.hasItem(UnnecessaryLocalRuleTest.RULE))
85          );
86      }
87  
88      private List<String> violations(final String file) throws Exception {
89          final String name = String.format(UnnecessaryLocalRuleTest.PATH, file);
90          final Environment env = new Environment.Mock().withFile(
91              name,
92              new TextOf(this.getClass().getResourceAsStream(file)).asString()
93          );
94          return new PmdValidator(env).validate(
95              Collections.singletonList(new File(env.basedir(), name))
96          ).stream().map(Violation::name).collect(Collectors.toList());
97      }
98  }
