5   package com.qulice.checkstyle;
6   
7   import com.google.common.collect.FluentIterable;
8   import java.util.ArrayList;
9   import java.util.Collection;
10  import java.util.Iterator;
11  
12  /**
13   * Represents a set of LineRange objects. For example, an instance of this class
14   * could represent all the line ranges for methods in a given Java source code
15   * file.
16   * @since 0.16
17   */
18  public final class LineRanges {
19  
20      /**
21       * Line ranges collected so far.
22       */
23      private final Collection<LineRange> lines;
24  
25      /**
26       * Default constructor.
27       */
28      public LineRanges() {
29          this.lines = new ArrayList<>(20);
30      }
31  
32      /**
33       * Adds a line range to the collection.
34       * @param line The line range to add to the collection
35       */
36      public void add(final LineRange line) {
37          this.lines.add(line);
38      }
39  
40      /**
41       * Returns an iterator for this collection.
42       * @return Iterator pointing to the internal collections elements
43       */
44      public Iterator<LineRange> iterator() {
45          return this.lines.iterator();
46      }
47  
48      /**
49       * Detects if the given line number is within any of the line ranges.
50       * @param line The given line number to check
51       * @return True if the given line number is within any line range
52       */
53      public boolean inRange(final int line) {
54          return !this.lines.isEmpty()
55              && FluentIterable.from(this.lines)
56              .anyMatch(new LineWithAny(line));
57      }
58  
59      /**
60       * Gets the subset of LineRanges that are within all given ranges. Does
61       * not return null; instead, returns empty range if there are no matches.
62       * @param ranges The ranges to filter on
63       * @return Returns all LineRange elements that are within range
64       */
65      public LineRanges within(final LineRanges ranges) {
66          final LineRanges result = new com.qulice.checkstyle.LineRanges();
67          final Iterator<LineRange> iterator = ranges.iterator();
68          while (iterator.hasNext()) {
69              final LineRange next = iterator.next();
70              for (final LineRange line : this.lines) {
71                  if (next.within(line)) {
72                      result.add(line);
73                  }
74              }
75          }
76          return result;
77      }
78  
79      /**
80       * Clears the collection.
81       */
82      public void clear() {
83          this.lines.clear();
84      }
85  }
