5   package com.qulice.checkstyle;
6   
7   import com.puppycrawl.tools.checkstyle.api.DetailAST;
8   import com.puppycrawl.tools.checkstyle.api.TokenTypes;
9   
10  /**
11   * Number of parameters, tolerant to constructors that only take attributes
12   * and to private methods.
13   *
14   * <p>The stock {@code ParameterNumber} check treats a constructor as it
15   * treats a method, while a constructor has no choice: it must accept a
16   * value for every attribute of its class. A long list of parameters there
17   * is a symptom of a class with too many attributes, and the class is what
18   * has to be blamed, not its constructor. That's why a constructor with no
19   * more parameters than the number of attributes of its class passes here,
20   * no matter how big the limit is.
21   *
22   * <p>A private method passes too. It is invisible outside its class, its
23   * parameters belong to no contract anybody else can see, and wrapping them
24   * into a new type only to satisfy the limit buys nothing.
25   *
26   * <p>So does a method of a JNA binding, whose parameter list repeats the
27   * signature of a native function and cannot be shortened without breaking
28   * the mapping.
29   *
30   * @since 1.0
31   */
32  public final class ParameterNumberCheck
33      extends com.puppycrawl.tools.checkstyle.checks.sizes.ParameterNumberCheck {
34  
35      /**
36       * Default constructor.
37       */
38      public ParameterNumberCheck() {
39          // nothing to initialize
40      }
41  
42      @Override
43      public void visitToken(final DetailAST ast) {
44          if (!ParameterNumberCheck.tolerated(ast)) {
45              super.visitToken(ast);
46          }
47      }
48  
49      /**
50       * May this node keep its long list of parameters?
51       *
52       * <p>A constructor may, when it takes no more parameters than the
53       * number of attributes of its class. A method may, when it is
54       * private or when it belongs to a JNA binding.</p>
55       *
56       * @param ast The CTOR_DEF or METHOD_DEF node
57       * @return TRUE if the check has to stay silent about it
58       */
59      private static boolean tolerated(final DetailAST ast) {
60          final boolean answer;
61          if (ast.getType() == TokenTypes.CTOR_DEF) {
62              answer = ParameterNumberCheck.parameters(ast)
63                  <= ParameterNumberCheck.attributes(ast);
64          } else {
65              answer = ast.findFirstToken(TokenTypes.MODIFIERS)
66                  .findFirstToken(TokenTypes.LITERAL_PRIVATE) != null
67                  || new JnaBinding(ast).is();
68          }
69          return answer;
70      }
71  
72      /**
73       * How many parameters does this constructor take?
74       * @param ctor The CTOR_DEF node
75       * @return Number of parameters
76       */
77      private static int parameters(final DetailAST ctor) {
78          return ctor.findFirstToken(TokenTypes.PARAMETERS)
79              .getChildCount(TokenTypes.PARAMETER_DEF);
80      }
81  
82      /**
83       * How many attributes does the class of this constructor have?
84       *
85       * <p>Static fields stay out of the count, since they belong to the
86       * class and not to its objects. Components of a record count in, as
87       * they are the attributes the canonical constructor has to take.</p>
88       *
89       * @param ctor The CTOR_DEF node
90       * @return Number of attributes
91       */
92      private static long attributes(final DetailAST ctor) {
93          final DetailAST block = ctor.getParent();
94          return ParameterNumberCheck.components(block.getParent())
95              + new ChildStream(block).children().filter(
96                  node -> node.getType() == TokenTypes.VARIABLE_DEF
97                      && node.findFirstToken(TokenTypes.MODIFIERS)
98                          .findFirstToken(TokenTypes.LITERAL_STATIC) == null
99              ).count();
100     }
101 
102     /**
103      * How many components does this type declare, if it is a record?
104      * @param type The node of the type declaration
105      * @return Number of record components
106      */
107     private static int components(final DetailAST type) {
108         final DetailAST components = type.findFirstToken(TokenTypes.RECORD_COMPONENTS);
109         final int count;
110         if (components == null) {
111             count = 0;
112         } else {
113             count = components.getChildCount(TokenTypes.RECORD_COMPONENT_DEF);
114         }
115         return count;
116     }
117 }
