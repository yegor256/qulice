5   package com.qulice.checkstyle;
6   
7   import com.puppycrawl.tools.checkstyle.api.DetailAST;
8   import com.puppycrawl.tools.checkstyle.api.FullIdent;
9   import com.puppycrawl.tools.checkstyle.api.TokenTypes;
10  
11  /**
12   * A method of a JNA binding.
13   *
14   * <p>A type that extends {@code com.sun.jna.Library} transcribes a native
15   * library function by function. The name of every method and the length of
16   * its parameter list come from the native side, so its author has no say in
17   * either of them and the checks that judge them have nothing to say here.
18   *
19   * <p>Qulice runs Checkstyle on one file at a time, so {@code Library} is
20   * never resolved to a class. The name in the {@code extends} or
21   * {@code implements} clause is therefore trusted only when it is fully
22   * qualified or when the file imports it from {@code com.sun.jna}.
23   *
24   * @since 1.0
25   */
26  final class JnaBinding {
27  
28      /**
29       * Package of the JNA interface that marks a type as a native binding.
30       */
31      private static final String PACKAGE = "com.sun.jna";
32  
33      /**
34       * Simple name of that interface.
35       */
36      private static final String LIBRARY = "Library";
37  
38      /**
39       * Canonical name of that interface.
40       */
41      private static final String QUALIFIED =
42          JnaBinding.PACKAGE + '.' + JnaBinding.LIBRARY;
43  
44      /**
45       * The node of the method declaration.
46       */
47      private final DetailAST node;
48  
49      /**
50       * Ctor.
51       * @param method The METHOD_DEF node
52       */
53      JnaBinding(final DetailAST method) {
54          this.node = method;
55      }
56  
57      /**
58       * Is this method a part of a JNA binding?
59       * @return TRUE if the type that declares it maps a native library
60       */
61      boolean is() {
62          DetailAST type = this.node.getParent();
63          boolean answer = false;
64          while (type != null) {
65              if (type.getType() == TokenTypes.CLASS_DEF
66                  || type.getType() == TokenTypes.INTERFACE_DEF) {
67                  answer = JnaBinding.mapped(type);
68                  break;
69              }
70              type = type.getParent();
71          }
72          return answer;
73      }
74  
75      /**
76       * Does this type declare JNA's interface as its supertype?
77       * @param type The CLASS_DEF or INTERFACE_DEF node
78       * @return TRUE if it does
79       */
80      private static boolean mapped(final DetailAST type) {
81          return JnaBinding.declares(type, TokenTypes.EXTENDS_CLAUSE)
82              || JnaBinding.declares(type, TokenTypes.IMPLEMENTS_CLAUSE);
83      }
84  
85      /**
86       * Does this clause of the type name JNA's interface?
87       * @param type The CLASS_DEF or INTERFACE_DEF node
88       * @param clause The token of the clause to look into
89       * @return TRUE if it does
90       */
91      private static boolean declares(final DetailAST type, final int clause) {
92          final DetailAST names = type.findFirstToken(clause);
93          final boolean answer;
94          if (names == null) {
95              answer = false;
96          } else {
97              answer = new ChildStream(names).children()
98                  .anyMatch(JnaBinding::library);
99          }
100         return answer;
101     }
102 
103     /**
104      * Is this name the one of JNA's interface?
105      * @param name The IDENT or DOT node of a supertype
106      * @return TRUE if it is
107      */
108     private static boolean library(final DetailAST name) {
109         final String text = FullIdent.createFullIdent(name).getText();
110         return JnaBinding.QUALIFIED.equals(text)
111             || JnaBinding.LIBRARY.equals(text) && JnaBinding.imported(name);
112     }
113 
114     /**
115      * Does the file that holds this node import JNA's interface?
116      * @param node The node to start the walk from
117      * @return TRUE if it does
118      */
119     private static boolean imported(final DetailAST node) {
120         DetailAST root = node;
121         while (root.getParent() != null) {
122             root = root.getParent();
123         }
124         return new ChildStream(root).children().anyMatch(JnaBinding::importing);
125     }
126 
127     /**
128      * Does this node import JNA's interface?
129      * @param node The node of a top-level declaration
130      * @return TRUE if it does
131      */
132     private static boolean importing(final DetailAST node) {
133         return node.getType() == TokenTypes.IMPORT
134             && JnaBinding.jna(
135                 FullIdent.createFullIdentBelow(node).getText()
136             );
137     }
138 
139     /**
140      * Does this imported name bring JNA's interface in?
141      * @param text The name of the import
142      * @return TRUE if it does
143      */
144     private static boolean jna(final String text) {
145         return JnaBinding.QUALIFIED.equals(text)
146             || String.format("%s.*", JnaBinding.PACKAGE).equals(text);
147     }
148 }
