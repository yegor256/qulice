5   package com.qulice.checkstyle;
6   
7   import com.puppycrawl.tools.checkstyle.api.DetailAST;
8   import com.puppycrawl.tools.checkstyle.api.FullIdent;
9   import com.puppycrawl.tools.checkstyle.api.TokenTypes;
10  import java.util.Collection;
11  import java.util.Set;
12  
13  /**
14   * A method of a JNA binding.
15   *
16   * <p>A type that extends {@code com.sun.jna.Library}, or its Windows
17   * flavour {@code com.sun.jna.win32.StdCallLibrary}, transcribes a native
18   * library function by function. The name of every method and the length of
19   * its parameter list come from the native side, so its author has no say in
20   * either of them and the checks that judge them have nothing to say here.</p>
21   *
22   * <p>Qulice runs Checkstyle on one file at a time, so neither of the two
23   * interfaces is ever resolved to a class. The name in the {@code extends}
24   * or {@code implements} clause is therefore trusted only when it is fully
25   * qualified or when the file imports it from its own package.</p>
26   *
27   * @since 1.0
28   */
29  final class JnaBinding {
30  
31      /**
32       * Canonical names of the JNA interfaces that mark a type as a native
33       * binding.
34       */
35      private static final Collection<String> LIBRARIES = Set.of(
36          "com.sun.jna.Library",
37          "com.sun.jna.win32.StdCallLibrary"
38      );
39  
40      /**
41       * The node of the method declaration.
42       */
43      private final DetailAST node;
44  
45      /**
46       * Ctor.
47       * @param method The METHOD_DEF node
48       */
49      JnaBinding(final DetailAST method) {
50          this.node = method;
51      }
52  
53      /**
54       * Is this method a part of a JNA binding?
55       * @return TRUE if the type that declares it maps a native library
56       */
57      boolean is() {
58          DetailAST type = this.node.getParent();
59          boolean answer = false;
60          while (type != null) {
61              if (type.getType() == TokenTypes.CLASS_DEF
62                  || type.getType() == TokenTypes.INTERFACE_DEF) {
63                  answer = JnaBinding.mapped(type);
64                  break;
65              }
66              type = type.getParent();
67          }
68          return answer;
69      }
70  
71      private static boolean mapped(final DetailAST type) {
72          return JnaBinding.declares(type, TokenTypes.EXTENDS_CLAUSE)
73              || JnaBinding.declares(type, TokenTypes.IMPLEMENTS_CLAUSE);
74      }
75  
76      private static boolean declares(final DetailAST type, final int clause) {
77          final DetailAST names = type.findFirstToken(clause);
78          final boolean answer;
79          if (names == null) {
80              answer = false;
81          } else {
82              answer = new ChildStream(names).children()
83                  .anyMatch(JnaBinding::library);
84          }
85          return answer;
86      }
87  
88      private static boolean library(final DetailAST name) {
89          final String text = FullIdent.createFullIdent(name).getText();
90          return JnaBinding.LIBRARIES.stream().anyMatch(
91              known -> known.equals(text)
92                  || JnaBinding.simple(known).equals(text)
93                  && JnaBinding.imported(name, known)
94          );
95      }
96  
97      private static boolean imported(final DetailAST node, final String known) {
98          DetailAST root = node;
99          while (root.getParent() != null) {
100             root = root.getParent();
101         }
102         return new ChildStream(root).children().anyMatch(
103             child -> JnaBinding.importing(child, known)
104         );
105     }
106 
107     private static boolean importing(final DetailAST node, final String known) {
108         return node.getType() == TokenTypes.IMPORT
109             && JnaBinding.brings(
110                 FullIdent.createFullIdentBelow(node).getText(), known
111             );
112     }
113 
114     private static boolean brings(final String text, final String known) {
115         return known.equals(text)
116             || String.format("%s.*", JnaBinding.pack(known)).equals(text);
117     }
118 
119     private static String simple(final String known) {
120         return known.substring(known.lastIndexOf('.') + 1);
121     }
122 
123     private static String pack(final String known) {
124         return known.substring(0, known.lastIndexOf('.'));
125     }
126 }
