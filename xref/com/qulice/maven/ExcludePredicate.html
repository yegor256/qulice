5   package com.qulice.maven;
6   
7   import com.google.common.base.Predicate;
8   import java.util.Collection;
9   
10  /**
11   * Predicate for excluded dependencies.
12   * @since 0.1
13   */
14  final class ExcludePredicate implements Predicate<String> {
15  
16      /**
17       * List of excludes.
18       */
19      private final Collection<String> excludes;
20  
21      /**
22       * Constructor.
23       * @param excludes List of excludes
24       */
25      ExcludePredicate(final Collection<String> excludes) {
26          this.excludes = excludes;
27      }
28  
29      @Override
30      public boolean apply(final String name) {
31          boolean ignore = false;
32          for (final String exclude : this.excludes) {
33              if (name.startsWith(exclude)) {
34                  ignore = true;
35                  break;
36              }
37          }
38          return ignore;
39      }
40  }
