5   package com.qulice.pmd.rules;
6   
7   import net.sourceforge.pmd.lang.java.ast.ASTExpression;
8   import net.sourceforge.pmd.lang.java.ast.ASTInfixExpression;
9   import net.sourceforge.pmd.lang.java.ast.ASTMethodCall;
10  import net.sourceforge.pmd.lang.java.ast.ASTNumericLiteral;
11  import net.sourceforge.pmd.lang.java.ast.BinaryOp;
12  import net.sourceforge.pmd.lang.java.rule.AbstractJavaRulechainRule;
13  import net.sourceforge.pmd.lang.java.types.JTypeMirror;
14  
15  /**
16   * Rule to prohibit use of String.length() when checking for empty string.
17   * String.isEmpty() should be used instead.
18   *
19   * <p>Only comparisons that have an exact {@code isEmpty()} equivalent are
20   * flagged. With {@code length()} on the left, those are {@code == 0},
21   * {@code != 0}, {@code > 0}, {@code <= 0}, {@code < 1} and {@code >= 1};
22   * the same six are recognised when the operands are swapped. Comparisons
23   * against {@code 1} such as {@code length() == 1} or {@code length() > 1}
24   * describe single-character strings and have no {@code isEmpty()}
25   * counterpart, so they are left alone.
26   *
27   * @since 0.18
28   */
29  public final class UseStringIsEmptyRule extends AbstractJavaRulechainRule {
30  
31      public UseStringIsEmptyRule() {
32          super(ASTInfixExpression.class);
33      }
34  
35      @Override
36      public Object visit(final ASTInfixExpression expr, final Object data) {
37          final ASTExpression left = expr.getLeftOperand();
38          final ASTExpression right = expr.getRightOperand();
39          if (isEmptyEquivalent(expr.getOperator(), left, right)
40              || isEmptyEquivalent(mirror(expr.getOperator()), right, left)
41          ) {
42              asCtx(data).addViolation(expr);
43          }
44          return data;
45      }
46  
47      /**
48       * Is this an {@code isEmpty()}-equivalent comparison, assuming the
49       * {@code length()} call is the left operand and the literal is the right?
50       * @param operator The comparison operator
51       * @param length The candidate {@code String.length()} call
52       * @param literal The candidate numeric literal
53       * @return True if the comparison can be rewritten with {@code isEmpty()}
54       */
55      private static boolean isEmptyEquivalent(
56          final BinaryOp operator,
57          final ASTExpression length,
58          final ASTExpression literal
59      ) {
60          return isStringLength(length) && isWhitelisted(operator, literal);
61      }
62  
63      /**
64       * Is the pair of operator and literal one of the six allowed
65       * {@code isEmpty()} equivalents ({@code == 0}, {@code != 0}, {@code > 0},
66       * {@code <= 0}, {@code < 1}, {@code >= 1})?
67       * @param operator The comparison operator
68       * @param literal The candidate numeric literal
69       * @return True if the pairing matches the whitelist
70       */
71      private static boolean isWhitelisted(
72          final BinaryOp operator,
73          final ASTExpression literal
74      ) {
75          boolean result = false;
76          if (literal instanceof ASTNumericLiteral lit) {
77              final String image = lit.getImage();
78              result = switch (operator) {
79                  case EQ, NE, GT, LE -> "0".equals(image);
80                  case LT, GE -> "1".equals(image);
81                  default -> false;
82              };
83          }
84          return result;
85      }
86  
87      /**
88       * Mirror an operator so a {@code literal OP length()} comparison can be
89       * evaluated as if {@code length()} were on the left.
90       * @param operator The operator as written
91       * @return The operator with its operands swapped
92       */
93      private static BinaryOp mirror(final BinaryOp operator) {
94          return switch (operator) {
95              case GT -> BinaryOp.LT;
96              case LT -> BinaryOp.GT;
97              case GE -> BinaryOp.LE;
98              case LE -> BinaryOp.GE;
99              default -> operator;
100         };
101     }
102 
103     /**
104      * Is the expression a no-argument {@code length()} call on a
105      * {@link String}?
106      * @param expr The expression to check
107      * @return True if it is a {@code String.length()} call
108      */
109     private static boolean isStringLength(final ASTExpression expr) {
110         boolean result = false;
111         if (expr instanceof ASTMethodCall call && call.getQualifier() != null) {
112             result = "length".equals(call.getMethodName())
113                 && call.getArguments().isEmpty()
114                 && isStringExpression(call.getQualifier());
115         }
116         return result;
117     }
118 
119     /**
120      * Checks if the expression is of {@link String} type.
121      * @param expr The expression to check
122      * @return True if expression has {@link String} type
123      */
124     private static boolean isStringExpression(final ASTExpression expr) {
125         final JTypeMirror type = expr.getTypeMirror();
126         return type.isClassOrInterface()
127             && String.class.getName().equals(type.toString());
128     }
129 }
