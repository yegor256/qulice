5   package com.qulice.pmd.rules;
6   
7   import java.util.Map;
8   import java.util.Set;
9   import net.sourceforge.pmd.lang.ast.Node;
10  import net.sourceforge.pmd.lang.java.ast.ASTAmbiguousName;
11  import net.sourceforge.pmd.lang.java.ast.ASTBlock;
12  import net.sourceforge.pmd.lang.java.ast.ASTClassType;
13  import net.sourceforge.pmd.lang.java.ast.ASTConstructorCall;
14  import net.sourceforge.pmd.lang.java.ast.ASTExpression;
15  import net.sourceforge.pmd.lang.java.ast.ASTMethodCall;
16  import net.sourceforge.pmd.lang.java.ast.ASTTypeExpression;
17  import net.sourceforge.pmd.lang.java.ast.ASTVariableAccess;
18  import net.sourceforge.pmd.lang.java.ast.ASTVariableDeclarator;
19  
20  /**
21   * Heuristics that veto an {@link UnnecessaryLocalRule} report when the local
22   * is semantically required (closure capture, clock snapshot, or capture
23   * before a destructive call). See issue #1607.
24   * @since 0.27.7
25   */
26  final class UnnecessaryLocalSkips {
27  
28      /**
29       * Simple class names from {@code java.time} whose {@code now()} factory
30       * returns a fresh snapshot of the system clock.
31       */
32      private static final Set<String> TIME_TYPES = Set.of(
33          "Instant", "LocalDate", "LocalDateTime", "LocalTime",
34          "ZonedDateTime", "OffsetDateTime", "OffsetTime",
35          "Year", "YearMonth", "MonthDay", "Clock"
36      );
37  
38      /**
39       * Other (non-{@code now()}) calls that return fresh state on each
40       * invocation, keyed by the qualifier's simple name.
41       */
42      private static final Map<String, Set<String>> FRESH_STATE_CALLS = Map.of(
43          "System", Set.of("currentTimeMillis", "nanoTime"),
44          "UUID", Set.of("randomUUID"),
45          "Math", Set.of("random")
46      );
47  
48      private UnnecessaryLocalSkips() {
49      }
50  
51      /**
52       * The single use of the local is reachable only by crossing a lambda or
53       * anonymous-class boundary, so the local exists to carry the value into
54       * a different exception scope.
55       * @param block The block enclosing the declaration
56       * @param name The variable name
57       * @param crossings Number of usages found when crossing find boundaries
58       * @return True when at least one usage sits behind a find boundary
59       */
60      static boolean acrossBoundary(
61          final ASTBlock block,
62          final String name,
63          final int crossings
64      ) {
65          return crossings != block
66              .descendants(ASTVariableAccess.class)
67              .filter(ref -> name.equals(ref.getName()))
68              .count();
69      }
70  
71      /**
72       * The initialiser snapshots mutable global state - a clock, a randomness
73       * source, or {@code new Date()} - so inlining would change <em>when</em>
74       * the value is taken.
75       * @param init The initialiser expression
76       * @return True if the initialiser captures fresh state
77       */
78      static boolean freshState(final ASTExpression init) {
79          final boolean result;
80          if (init instanceof ASTMethodCall call) {
81              result = UnnecessaryLocalSkips.freshStateCall(call);
82          } else if (init instanceof ASTConstructorCall call) {
83              final ASTClassType type = call.getTypeNode();
84              result = type != null && "Date".equals(type.getSimpleName());
85          } else {
86              result = false;
87          }
88          return result;
89      }
90  
91      /**
92       * The initialiser is a method call on some target, and a later statement
93       * (before the local is read) calls a method on that same target. Inlining
94       * would read the post-mutation value.
95       * @param variable The variable declarator
96       * @param use The single use of the variable
97       * @return True if an intervening statement calls the initialiser's target
98       */
99      static boolean interveningCall(
100         final ASTVariableDeclarator variable,
101         final ASTVariableAccess use
102     ) {
103         boolean found = false;
104         final ASTExpression init = variable.getInitializer();
105         if (init instanceof ASTMethodCall call) {
106             final String target = UnnecessaryLocalSkips.qualifierImage(
107                 call.getQualifier()
108             );
109             if (target != null && !target.isEmpty()) {
110                 final Node decl = UnnecessaryLocalSkips.blockLevel(variable);
111                 final Node consumer = UnnecessaryLocalSkips.blockLevel(use);
112                 found = UnnecessaryLocalSkips.sameBlock(decl, consumer)
113                     && UnnecessaryLocalSkips.scanBetween(decl, consumer, target);
114             }
115         }
116         return found;
117     }
118 
119     private static boolean sameBlock(final Node first, final Node second) {
120         return first != null && second != null
121             && first.getParent() instanceof ASTBlock
122             && first.getParent().equals(second.getParent());
123     }
124 
125     private static boolean freshStateCall(final ASTMethodCall call) {
126         final String name = call.getMethodName();
127         final String qualifier = UnnecessaryLocalSkips.qualifierImage(
128             call.getQualifier()
129         );
130         boolean fresh = false;
131         if (qualifier != null) {
132             final boolean known = UnnecessaryLocalSkips.FRESH_STATE_CALLS
133                 .getOrDefault(qualifier, Set.of()).contains(name);
134             final boolean clock = "now".equals(name)
135                 && UnnecessaryLocalSkips.TIME_TYPES.contains(qualifier);
136             fresh = known || clock;
137         }
138         return fresh;
139     }
140 
141     private static boolean scanBetween(
142         final Node decl,
143         final Node consumer,
144         final String target
145     ) {
146         final Node block = decl.getParent();
147         final int last = consumer.getIndexInParent();
148         boolean found = false;
149         for (int idx = decl.getIndexInParent() + 1; idx < last && !found;
150             idx += 1) {
151             found = block.getChild(idx).descendants(ASTMethodCall.class)
152                 .crossFindBoundaries()
153                 .toStream()
154                 .anyMatch(call -> UnnecessaryLocalSkips.callsTarget(call, target));
155         }
156         return found;
157     }
158 
159     private static boolean callsTarget(
160         final ASTMethodCall call,
161         final String target
162     ) {
163         return target.equals(
164             UnnecessaryLocalSkips.qualifierImage(call.getQualifier())
165         );
166     }
167 
168     private static Node blockLevel(final Node node) {
169         Node current = node;
170         while (current != null && !(current.getParent() instanceof ASTBlock)) {
171             current = current.getParent();
172         }
173         return current;
174     }
175 
176     private static String qualifierImage(final ASTExpression expr) {
177         final String result;
178         if (expr instanceof ASTAmbiguousName name) {
179             result = name.getName();
180         } else if (expr instanceof ASTVariableAccess access) {
181             result = access.getName();
182         } else if (expr instanceof ASTTypeExpression type
183             && type.getTypeNode() instanceof ASTClassType klass) {
184             result = klass.getSimpleName();
185         } else {
186             result = null;
187         }
188         return result;
189     }
190 }
